# 开局抽卡纯前端版 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 StoryOS 开局设计做成纯前端抽卡页面（零 LLM、零依赖），两文件部署到 atposs.com。

**Architecture:** Python 导出脚本把 315 题材 YAML + 31 幕结构 + 10 世界观 + 姓名库瘦身成 `web/gacha/data.js`；`web/gacha/index.html` 内嵌种子驱动抽卡引擎（mulberry32），5 区块独立子种子支持部分锁定，开局码 = 5 段 base36 拼接可还原。

**Tech Stack:** Python 3.11 + PyYAML（项目已有）、原生 HTML/CSS/JS、pytest。

**Spec:** `docs/superpowers/specs/2026-10-07-gacha-web-design.md`

## Global Constraints

- 页面零依赖、零网络请求；不准引入任何 JS 库/CDN
- 导出必须用项目 venv：`.venv/Scripts/python.exe`（Windows）
- data.js 是部署产物，**要提交 git**
- 世界观的「骨架选择」步骤语义 ≠ 本项目的幕结构模板：本计划中「骨架」= macro 幕结构（31 个，排除 `custom`/`ai_custom`）
- 编码一律 UTF-8；JSON 输出 `ensure_ascii=False`
- 风格沿用 StoryOS 前端纸质书卷风（米色底 `#f7f3ea`、卡片 `#fffdf8`、主按钮橙棕 `#b5501f`、衬线标题）

  ```

"""开局抽卡导出脚本测试（spec: docs/superpowers/specs/2026-10-07-gacha-web-design.md）"""
import json
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
DATA_JS = ROOT / "web" / "gacha" / "data.js"


def _run_export() -> dict:
    subprocess.run(
        [sys.executable, str(ROOT / "scripts" / "export_gacha_web.py")],
        check=True, cwd=ROOT, capture_output=True, text=True,
    )
    text = DATA_JS.read_text(encoding="utf-8")
    m = re.fullmatch(r"window\.GACHA_DATA = (\{.*\});\n?", text, re.S)
    assert m, "data.js 必须是单行 window.GACHA_DATA = {...};"
    return json.loads(m.group(1))


def test_export_integrity():
    data = _run_export()
    yaml_count = len(list((ROOT / "story_engine" / "plugins" / "genres").glob("*.yaml")))
    assert len(data["genres"]) == yaml_count, "题材数必须等于 YAML 文件数"
    for g in data["genres"]:
        assert g["title"] and g["coreConflict"] and g["tracks"], g["id"]
        assert g["setting"] and g["characters"], g["id"]
    assert len(data["worldviews"]) == 10
    for w in data["worldviews"]:
        assert len(w["rules"]) >= 6, w["key"]
    assert len(data["skeletons"]) >= 30
    for s in data["skeletons"]:
        eps = [int(b["ep"]) for a in s["acts"] for b in a["beats"]]
        assert eps and all(1 <= e <= 12 for e in eps), s["id"]
    assert set(data["namePools"]) == {"cn", "jp", "western", "other"}
    assert "jianghu-martial" in data["culturePool"]
```

def _value_label(param_key: str, value: str) -> str:
    param = ALL_PARAMS.get(param_key) or {}
    for opt in param.get("options", []):
        if opt.get("value") == value:
            return opt.get("label", value)
    return value


def export_worldviews() -> list[dict]:
    out = []
    for p in PRESETS:
        rules = []
        for key in RULE_PARAM_KEYS:
            val = p["params"].get(key)
            if val is None:
                continue
            rules.append(f"{ALL_PARAMS[key]['label']}：{_value_label(key, val)}")
        out.append({"key": p["key"], "name": p["name"],
                    "vibe": p["vibe"], "rules": rules})
    return out


def export_skeletons() -> list[dict]:
    out = []
    for name in TEMPLATES:
        if name in ("custom", "ai_custom"):
            continue
        cn, brief = SKELETON_NAMES.get(name, (name, ""))
        st = compute_acts(name, TOTAL_EPISODES)
        acts = [{
            "name": a.name, "function": a.function, "range": a.episode_range,
            "beats": [{"name": b.name, "ep": b.ep, "desc": b.desc}
                      for b in a.beats],
        } for a in st.acts]
        out.append({"id": name, "name": cn, "brief": brief, "acts": acts})
    return out


def export_genres() -> list[dict]:
    out = []
    for f in sorted(GENRES_DIR.glob("*.yaml")):
        raw = yaml.safe_load(f.read_text(encoding="utf-8"))
        params = raw.get("params") or {}
        prompt = params.get("prompt") or {}
        fusion = raw.get("fusion") or {}
        gid = raw["name"]
        title = params.get("title")
        core = fusion.get("core_conflict")
        tracks = [{"id": t["id"], "name": t["name"]}
                  for t in params.get("tracks") or []]
        if not title or not core or not tracks:
            raise SystemExit(f"题材 {gid} 缺 title/core_conflict/tracks，导出中止")
        tags = params.get("taxonomy_tags") or fusion.get("parent_genres") or []
        conflict_words = [CONFLICT_WORDS.get(c["type"], c["type"])
                          for c in params.get("conflict_types") or []]
        out.append({
            "id": gid,
            "title": title,
            "tags": tags,
            "coreConflict": core,
            "setting": prompt.get("setting", ""),
            "characters": prompt.get("characters", ""),
            "role": prompt.get("role", ""),
            "tracks": tracks,
            "conflictWords": conflict_words,
            "resolution": params.get("resolution_pattern", ""),
            "recommendedPreset": params.get("recommended_preset"),
            "recommendedCulture": params.get("recommended_culture"),
            "recommendedTemplates": list(macro_templates_for_genre(gid)),
        })
    return out


def main() -> None:
    data = {
        "version": datetime.date.today().isoformat(),
        "totalEpisodes": TOTAL_EPISODES,
        "genres": export_genres(),
        "worldviews": export_worldviews(),
        "skeletons": export_skeletons(),
        "namePools": NAME_POOLS,
        "culturePool": CULTURE_POOL,
    }
    OUT_FILE.parent.mkdir(parents=True, exist_ok=True)
    OUT_FILE.write_text(
        "window.GACHA_DATA = "
        + json.dumps(data, ensure_ascii=False, separators=(",", ":")) + ";\n",
        encoding="utf-8",
    )
    size_kb = OUT_FILE.stat().st_size // 1024
    print(f"导出完成：{OUT_FILE.name}  题材 {len(data['genres'])}  "
          f"世界观 {len(data['worldviews'])}  骨架 {len(data['skeletons'])}  "
          f"体积 {size_kb}KB")


if __name__ == "__main__":
    main()

function drawNames(pool, rng) {
  const mk = () => {
    if (pool.names) return pick(pool.names, rng);
    if (pool.family) return pick(pool.given, rng) + "·" + pick(pool.family, rng);
    return pick(pool.surnames, rng) + pick(pool.given, rng);
  };
  const a = mk(); let b = mk(), c = mk();
  while (b === a) b = mk();
  while (c === a || c === b) c = mk();
  return {mc: a, rival: b, ally: c};
}

function fill(tpl, g, w, names, rng) {
  const conflict = g.conflictWords.length ? pick(g.conflictWords, rng) : "矛盾";
  return tpl
    .replaceAll("{mc}", names.mc).replaceAll("{rival}", names.rival)
    .replaceAll("{ally}", names.ally)
    .replaceAll("{main}", g.tracks[0] ? g.tracks[0].name : "主线")
    .replaceAll("{sub}", g.tracks[1] ? g.tracks[1].name : "副线")
    .replaceAll("{conflict}", conflict)
    .replaceAll("{world}", w.name)
    .replaceAll("{res}", g.resolution || "主题兑现");
}

function beatLine(beat, g, w, names, rng) {
  const text = beat.name + " " + beat.desc;
  let cls = "generic";
  for (const [re, c] of BEAT_RULES) if (re.test(text)) { cls = c; break; }
  return fill(pick(SENTENCES[cls], rng), g, w, names, rng);
}

function buildModel(seeds) {
  const D = window.GACHA_DATA;
  const g = pick(D.genres, rngOf(seeds.genre));
  const wr = rngOf(seeds.worldview);
  let w = (g.recommendedPreset && wr() < 0.7)
    ? (D.worldviews.find(x => x.key === g.recommendedPreset) || pick(D.worldviews, wr))
    : pick(D.worldviews, wr);
  const sr = rngOf(seeds.skeleton);
  let s = (g.recommendedTemplates.length && sr() < 0.7)
    ? (D.skeletons.find(x => x.id === g.recommendedTemplates[0]) || pick(D.skeletons, sr))
    : pick(D.skeletons, sr);
  const cr = rngOf(seeds.cast);
  const culture = g.recommendedCulture || "modern-chinese-urban";
  const pool = D.namePools[D.culturePool[culture] || "cn"];
  const names = drawNames(pool, cr);
  const orng = rngOf(seeds.outline);
  const acts = s.acts.map(a => ({

    beats: a.beats.map(b => ({ep: b.ep, desc: b.desc, line: beatLine(b, g, w, names, orng)})),
  }));
  return {g, w, s, culture, names, acts, hook: fill(pick(HOOKS, orng), g, w, names, orng)};
}

function decode(code) {

}

}
